    <!-- SCRIPT DE LÓGICA CON LOCALSTORAGE -->
    <script>
        // Cargar datos persistentes o valores por defecto
        let routines = JSON.parse(localStorage.getItem('tititeam_routines')) || [
            {
                id: 1,
                name: "Día 1: Tren Superior (Empuje)",
                exercises: [
                    { name: "extensión triceps", category: "Tríceps", sets: 3, reps: "10-12", rest: "90s", notes: "Mantener codos fijos.", img: "https://lh3.googleusercontent.com/d/1N66CAvEnTuPAqOLaclNvm5hKHQNpRORu" }
                ]
            }
        ];

        let activeRoutineId = parseInt(localStorage.getItem('tititeam_active_routine')) || (routines.length > 0 ? routines[0].id : 1);
        let trainingHistory = JSON.parse(localStorage.getItem('tititeam_history')) || [];
        let lastPersonalRecords = JSON.parse(localStorage.getItem('tititeam_records')) || {};

        function saveDataToLocalStorage() {
            localStorage.setItem('tititeam_routines', JSON.stringify(routines));
            localStorage.setItem('tititeam_active_routine', activeRoutineId);
            localStorage.setItem('tititeam_history', JSON.stringify(trainingHistory));
            localStorage.setItem('tititeam_records', JSON.stringify(lastPersonalRecords));
        }

        function updatePreview() {
            const select = document.getElementById('catalogSelect');
            if (!select || select.selectedIndex === -1) return;
            const opt = select.options[select.selectedIndex];
            const imgEl = document.getElementById('previewImg');
            const catEl = document.getElementById('previewCategory');
            
            if (imgEl && opt) imgEl.src = opt.getAttribute('data-img') || '';
            if (catEl && opt) catEl.textContent = opt.getAttribute('data-cat') || '';
        }

        function switchView(view) {
            const trainerView = document.getElementById('trainerView');
            const studentView = document.getElementById('studentView');
            const historyView = document.getElementById('historyView');
            
            const btnTrainer = document.getElementById('btnTrainer');
            const btnStudent = document.getElementById('btnStudent');
            const btnHistory = document.getElementById('btnHistory');

            trainerView.classList.add('hidden');
            studentView.classList.add('hidden');
            historyView.classList.add('hidden');

            btnTrainer.className = "px-3.5 py-2 rounded-xl text-xs font-bold transition bg-slate-100 text-slate-600 hover:bg-slate-200 flex items-center gap-1.5 cursor-pointer";
            btnStudent.className = "px-3.5 py-2 rounded-xl text-xs font-bold transition bg-slate-100 text-slate-600 hover:bg-slate-200 flex items-center gap-1.5 cursor-pointer";
            btnHistory.className = "px-3.5 py-2 rounded-xl text-xs font-bold transition bg-slate-100 text-slate-600 hover:bg-slate-200 flex items-center gap-1.5 cursor-pointer";

            if (view === 'trainer') {
                trainerView.classList.remove('hidden');
                btnTrainer.className = "px-3.5 py-2 rounded-xl text-xs font-bold transition bg-purple-600 text-white shadow-sm flex items-center gap-1.5 cursor-pointer";
                renderTrainerRoutines();
                updatePreview();
            } else if (view === 'student') {
                studentView.classList.remove('hidden');
                btnStudent.className = "px-3.5 py-2 rounded-xl text-xs font-bold transition bg-emerald-600 text-white shadow-sm flex items-center gap-1.5 cursor-pointer";
                updateStudentSelect();
            } else if (view === 'history') {
                historyView.classList.remove('hidden');
                btnHistory.className = "px-3.5 py-2 rounded-xl text-xs font-bold transition bg-blue-600 text-white shadow-sm flex items-center gap-1.5 cursor-pointer";
                renderHistory();
            }
        }

        function createNewRoutine() {
            const nameInput = document.getElementById('routineNameInput');
            if (!nameInput || !nameInput.value.trim()) {
                alert("Escribe un nombre para la rutina.");
                return;
            }
            const newRoutine = {
                id: Date.now(),
                name: nameInput.value.trim(),
                exercises: []
            };
            routines.push(newRoutine);
            activeRoutineId = newRoutine.id;
            nameInput.value = "";
            saveDataToLocalStorage();
            renderTrainerRoutines();
            alert("¡Rutina creada con éxito!");
        }

        function deleteRoutine(routineId) {
            if (routines.length <= 1) {
                alert("Debes mantener al menos una rutina en el sistema.");
                return;
            }
            if (confirm("¿Estás seguro de que deseas eliminar esta rutina por completo?")) {
                routines = routines.filter(r => r.id !== routineId);
                if (activeRoutineId === routineId) {
                    activeRoutineId = routines[0].id;
                }
                saveDataToLocalStorage();
                renderTrainerRoutines();
            }
        }

        function deleteExerciseFromRoutine(routineId, exerciseIndex) {
            const routine = routines.find(r => r.id === routineId);
            if (routine) {
                routine.exercises.splice(exerciseIndex, 1);
                saveDataToLocalStorage();
                renderTrainerRoutines();
            }
        }

        function renderTrainerRoutines() {
            const list = document.getElementById('trainerRoutinesList');
            const label = document.getElementById('activeRoutineLabel');
            if (!list) return;
            list.innerHTML = "";

            if (routines.length === 0) {
                list.innerHTML = `<p class="text-xs text-slate-400">No hay rutinas creadas todavía.</p>`;
                if (label) label.textContent = "Ninguna";
                return;
            }

            if (!routines.some(r => r.id === activeRoutineId)) {
                activeRoutineId = routines[0].id;
            }

            const active = routines.find(r => r.id === activeRoutineId);
            if (label) label.textContent = active ? active.name : "Ninguna";

            routines.forEach(r => {
                const item = document.createElement('div');
                item.className = `p-4 rounded-xl border space-y-3 ${r.id === activeRoutineId ? 'bg-purple-50/40 border-purple-300 ring-1 ring-purple-300' : 'bg-white border-slate-200'}`;
                
                let exercisesHTML = '';
                if (r.exercises.length === 0) {
                    exercisesHTML = `<p class="text-xs text-slate-400 italic">No hay ejercicios agregados en esta rutina todavía.</p>`;
                } else {
                    r.exercises.forEach((ex, idx) => {
                        exercisesHTML += `
                            <div class="bg-white p-3 rounded-xl border border-slate-200 flex items-center justify-between gap-3">
                                <div class="flex items-center gap-3">
                                    <div class="w-12 h-12 bg-white rounded-lg border border-slate-200 p-1 flex items-center justify-center shrink-0">
                                        <img src="${ex.img}" alt="${ex.name}" class="w-full h-full object-contain">
                                    </div>
                                    <div>
                                        <span class="text-[9px] font-bold px-2 py-0.5 rounded bg-purple-50 text-purple-700">${ex.category}</span>
                                        <h4 class="text-xs font-bold text-slate-800">${ex.name}</h4>
                                        <span class="text-[10px] text-slate-500">Series: ${ex.sets} | Reps: ${ex.reps} | Descanso: ${ex.rest}</span>
                                    </div>
                                </div>
                                <button onclick="deleteExerciseFromRoutine(${r.id}, ${idx})" class="text-rose-500 hover:text-rose-700 p-1.5 cursor-pointer text-xs" title="Quitar ejercicio">
                                    <i class="ph-bold ph-x text-sm"></i>
                                </button>
                            </div>
                        `;
                    });
                }

                item.innerHTML = `
                    <div class="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-2 pb-2 border-b border-slate-100">
                        <div>
                            <strong class="text-sm font-bold text-slate-900">${r.name}</strong>
                            <span class="text-xs text-slate-500 ml-2">(${r.exercises.length} ejercicios)</span>
                        </div>
                        <div class="flex gap-2">
                            <button onclick="activeRoutineId = ${r.id}; saveDataToLocalStorage(); renderTrainerRoutines();" class="text-xs font-bold px-3 py-1.5 rounded-lg ${r.id === activeRoutineId ? 'bg-purple-600 text-white' : 'bg-purple-100 text-purple-700 hover:bg-purple-200'} cursor-pointer">
                                ${r.id === activeRoutineId ? 'Rutina Activa para Editar' : 'Seleccionar'}
                            </button>
                            <button onclick="deleteRoutine(${r.id})" class="text-xs font-bold px-3 py-1.5 rounded-lg bg-rose-50 text-rose-600 hover:bg-rose-100 cursor-pointer flex items-center gap-1">
                                <i class="ph-bold ph-trash"></i>
                            </button>
                        </div>
                    </div>
                    <div class="space-y-2">

                    </div>
                `;
                list.appendChild(item);
            });
        }

        function addExerciseToActiveRoutine() {
            const active = routines.find(r => r.id === activeRoutineId);
            if (!active) {
                alert("Selecciona o crea una rutina primero.");
                return;
            }

            const select = document.getElementById('catalogSelect');
            if (!select || select.selectedIndex === -1) return;
            const opt = select.options[select.selectedIndex];

            const setsCount = parseInt(document.getElementById('inputSets').value) || 3;

            active.exercises.push({
                name: opt.value,
                category: opt.getAttribute('data-cat'),
                sets: setsCount,
                reps: document.getElementById('inputReps').value,
                rest: document.getElementById('inputRest').value,
                notes: document.getElementById('inputNotes').value || "Sin notas adicionales.",
                img: opt.getAttribute('data-img')
            });

            document.getElementById('inputNotes').value = "";
            saveDataToLocalStorage();
            renderTrainerRoutines();
        }

        function updateStudentSelect() {
            const select = document.getElementById('studentRoutineSelect');
            if (!select) return;
            select.innerHTML = '<option value="">-- Selecciona una rutina --</option>';
            
            routines.forEach(r => {
                const opt = document.createElement('option');
                opt.value = r.id;
                opt.textContent = r.name;
                select.appendChild(opt);
            });
        }

        function loadStudentRoutine() {
            const routineId = document.getElementById('studentRoutineSelect').value;
            const container = document.getElementById('studentExercisesContainer');
            const generalSection = document.getElementById('studentGeneralImpressionsSection');
            if (!container) return;
            container.innerHTML = "";

            if (!routineId) {
                container.innerHTML = `<div class="bg-white p-8 rounded-2xl text-center text-slate-400 text-sm border border-slate-200">Por favor, selecciona una rutina válida en la parte superior.</div>`;
                if (generalSection) generalSection.classList.add('hidden');
                return;
            }

            const routine = routines.find(r => r.id == routineId);
            if (!routine || routine.exercises.length === 0) {
                container.innerHTML = `<div class="bg-white p-8 rounded-2xl text-center text-slate-400 text-sm border border-slate-200">Esta rutina no tiene ejercicios asignados todavía.</div>`;
                if (generalSection) generalSection.classList.add('hidden');
                return;
            }

            if (generalSection) generalSection.classList.remove('hidden');

            routine.exercises.forEach((ex) => {
                const lastRecord = lastPersonalRecords[ex.name];

                let setsInputsHTML = '';
                for (let i = 0; i < ex.sets; i++) {
                    const prevSeriesData = lastRecord && lastRecord[i] ? lastRecord[i] : { weight: '', reps: '' };
                    const placeholderWeight = prevSeriesData.weight ? `Anterior: ${prevSeriesData.weight}kg` : 'Ej. 50';
                    const placeholderReps = prevSeriesData.reps ? `Anterior: ${prevSeriesData.reps}` : 'Ej. 12';

                    setsInputsHTML += `
                        <div class="grid grid-cols-12 gap-2 items-center bg-white p-2 rounded-lg border border-emerald-100 set-row" data-set-index="${i}">
                            <div class="col-span-3 text-xs font-extrabold text-emerald-800">
                                Serie ${i + 1}
                            </div>
                            <div class="col-span-4">
                                <input type="number" placeholder="${placeholderWeight}" class="input-set-weight w-full bg-slate-50 border border-emerald-300 rounded px-2 py-1 text-xs font-bold text-slate-800" value="${prevSeriesData.weight}">
                            </div>
                            <div class="col-span-5">
                                <input type="text" placeholder="${placeholderReps}" class="input-set-reps w-full bg-slate-50 border border-emerald-300 rounded px-2 py-1 text-xs font-bold text-slate-800" value="${prevSeriesData.reps}">
                            </div>
                        </div>
                    `;
                }

                let lastSummaryText = 'Ningún registro previo';
                if (lastRecord && lastRecord.length > 0) {
                    lastSummaryText = lastRecord.map((s, idx) => `S${idx+1}: ${s.weight || 0}kg/${s.reps || '-'}r`).join(' | ');
                }

                const card = document.createElement('div');
                card.className = "bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden flex flex-col md:flex-row exercise-card-item";
                card.dataset.exerciseName = ex.name;

                card.innerHTML = `
                    <div class="p-5 md:w-72 bg-slate-50/50 border-b md:border-b-0 md:border-r border-slate-200 flex flex-col items-center justify-center text-center">
                        <div class="w-32 h-32 bg-white rounded-xl border border-slate-200 p-2 shadow-xs mb-3 flex items-center justify-center overflow-hidden">
                            <img src="${ex.img}" alt="${ex.name}" class="w-full h-full object-contain">
                        </div>
                        <span class="inline-block px-2.5 py-0.5 rounded-full text-[10px] font-bold bg-purple-50 text-purple-700 border border-purple-200 mb-1">
                            ${ex.category}
                        </span>
                        <h3 class="font-bold text-slate-900 text-base leading-snug">${ex.name}</h3>
                        
                        <div class="mt-2.5 bg-blue-50 border border-blue-200 rounded-lg p-1.5 w-full text-center">
                            <span class="block text-[9px] font-bold uppercase text-blue-600">Última Sesión</span>
                            <span class="text-[10px] font-extrabold text-blue-900 leading-tight block">${lastSummaryText}</span>
                        </div>
                    </div>

                    <div class="p-5 flex-1 flex flex-col justify-between gap-4">
                        <div class="bg-purple-50/60 border-l-4 border-purple-600 p-3.5 rounded-r-xl">
                            <span class="text-[11px] font-extrabold uppercase text-purple-700 tracking-wider block mb-1">Prescripción de GAUL (${ex.sets} Series)</span>
                            <div class="grid grid-cols-3 gap-2 text-center mb-2">
                                <div class="bg-white p-1.5 rounded border border-purple-100"><span class="block text-[10px] text-slate-400 uppercase">Series</span><strong class="text-slate-800 text-sm">${ex.sets}</strong></div>
                                <div class="bg-white p-1.5 rounded border border-purple-100"><span class="block text-[10px] text-slate-400 uppercase">Reps Obj</span><strong class="text-slate-800 text-sm">${ex.reps}</strong></div>
                                <div class="bg-white p-1.5 rounded border border-purple-100"><span class="block text-[10px] text-slate-400 uppercase">Descanso</span><strong class="text-slate-800 text-sm">${ex.rest}</strong></div>
                            </div>
                            <div class="text-xs text-slate-600"><em>"${ex.notes}"</em></div>
                        </div>

                        <div class="bg-emerald-50/60 border-l-4 border-emerald-600 p-3.5 rounded-r-xl space-y-2.5">
                            <div class="flex justify-between items-center">
                                <span class="text-[11px] font-extrabold uppercase text-emerald-700 tracking-wider">Registro por Series (Carga y Reps)</span>
                            </div>
                            
                            <div class="space-y-1.5 sets-container">
                                ${setsInputsHTML}
                            </div>

                                <label class="block text-[11px] font-semibold text-slate-600 mb-1">Impresiones del Ejercicio</label>

                        </div>
                    </div>

            });
        }

        function saveTrainingSession() {
            const routineId = document.getElementById('studentRoutineSelect').value;
            const routine = routines.find(r => r.id == routineId);
            if (!routine) return;

            const cards = document.querySelectorAll('.exercise-card-item');
            let sessionExercises = [];
            const currentDate = new Date().toLocaleDateString('es-ES', { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' });

            cards.forEach((card, idx) => {
                const exName = card.dataset.exerciseName;
                const impressions = card.querySelector('.input-impressions').value || 'Sin comentarios';
                const originalEx = routine.exercises[idx];

                const setRows = card.querySelectorAll('.set-row');
                let exerciseSetsData = [];

                setRows.forEach(row => {
                    const weight = row.querySelector('.input-set-weight').value || '0';
                    const reps = row.querySelector('.input-set-reps').value || '-';
                    exerciseSetsData.push({ weight, reps });
                });

                lastPersonalRecords[exName] = exerciseSetsData;

                sessionExercises.push({
                    name: exName,
                    category: originalEx.category,
                    setsData: exerciseSetsData,
                    impressions: impressions,
                    img: originalEx.img
                });
            });

            const generalImpressions = document.getElementById('generalImpressionsInput').value || 'Sin impresiones generales.';

            trainingHistory.unshift({
                date: currentDate,
                routineName: routine.name,
                generalImpressions: generalImpressions,
                exercises: sessionExercises
            });

            saveDataToLocalStorage();
            document.getElementById('generalImpressionsInput').value = "";
            alert("¡Entrenamiento guardado con éxito! Se han registrado tus series en el camino del éxito.");
            loadStudentRoutine();
        }

        function renderHistory() {
            const container = document.getElementById('historyContainer');
            if (!container) return;
            container.innerHTML = "";

            if (trainingHistory.length === 0) {
                container.innerHTML = `

                        Todavía no hay entrenamientos guardados. Completa una rutina en la pestaña "LeTITI" y pulsa en "Guardar Jornada Completa".
                    </div>
                `;
                return;
            }

            trainingHistory.forEach((session) => {
                const sessionCard = document.createElement('div');
                sessionCard.className = "bg-slate-50 p-5 rounded-2xl border border-slate-200 space-y-4";

                let exercisesHistoryHTML = '';

                    exercisesHistoryHTML += `

                            <div class="flex items-center justify-between gap-3">
                                <div class="flex items-center gap-3">

                        <strong>Impresiones Generales:</strong> <em>"${session.generalImpressions}"</em>
                    </div>
                `;
                container.appendChild(sessionCard);
            });
        }
